function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var timestamp = new Date();
  
  var nama = e.parameter.nama;
  var kelas = e.parameter.kelas;
  var mapel = e.parameter.mapel;
  var nilai = e.parameter.nilai;
  
  sheet.appendRow([timestamp, nama, kelas, mapel, nilai]);
  
  return ContentService
    .createTextOutput(JSON.stringify({ "result": "success", "message": "Data berhasil disimpan" }))
    .setMimeType(ContentService.MimeType.JSON);
}
